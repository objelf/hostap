    dev=$(readlink -f /sys/class/net/$i/device)

    echo "=== $i ==="
    ethtool -i "$i"
    echo "mac: $(cat /sys/class/net/$i/address)"
    echo "driver-link: $(readlink -f /sys/class/net/$i/device/driver)"
    echo "device-link: $dev"

    if [ -f "$dev/idVendor" ] && [ -f "$dev/idProduct" ]; then
        echo "vid:pid = $(cat "$dev/idVendor"):$(cat "$dev/idProduct")"
    else
        echo "vid:pid = not a direct USB device node, checking parents..."
        p="$dev"
        found=""
        while [ "$p" != "/" ]; do
            if [ -f "$p/idVendor" ] && [ -f "$p/idProduct" ]; then
                echo "vid:pid = $(cat "$p/idVendor"):$(cat "$p/idProduct")"
                found=1
                break
            fi
            p=$(dirname "$p")
        done
        [ -n "$found" ] || echo "vid:pid = not found"
    fi
    echo
done

sudo cat /sys/kernel/tracing/trace_pipe

for i in $(seq 1 10000); do
    echo "=== round $i ==="
    sudo ./wpa_cli -i wlp58s0 disconnect >/dev/null
    sleep 1
    sudo ./wpa_cli -i wlp58s0 reconnect >/dev/null

    state=""
    for t in $(seq 1 20); do
        state=$(sudo ./wpa_cli -i wlp58s0 status | awk -F= '/^wpa_state=/{print $2}')
        echo "state=$state"
        [ "$state" = "COMPLETED" ] && break
        sleep 1
    done

    if [ "$state" = "COMPLETED" ]; then
        echo "round $i connected, now disconnect"
        sudo ./wpa_cli -i wlp58s0 disconnect >/dev/null
    else
        echo "round $i timeout, final state=$state"
        sudo ./wpa_cli -i wlp58s0 disconnect >/dev/null
    fi

    sleep 1
done
